cmake_minimum_required(VERSION 3.25)
project(CompositorWindows VERSION 0.4.5 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(CTest)
include(FetchContent)
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Gui Widgets Concurrent Svg Test Network)
if(MSVC)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/bootstrap_fonts.py"
    RESULT_VARIABLE FONT_RESULT)
  if(NOT FONT_RESULT EQUAL 0)
    message(FATAL_ERROR "Could not install the Chinese fallback font")
  endif()
  file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/.tools/fonts" DESTINATION "${CMAKE_BINARY_DIR}")
  set(CMAKE_C_COMPILER_LAUNCHER "${Python3_EXECUTABLE};${CMAKE_CURRENT_SOURCE_DIR}/scripts/msvc.py")
  set(CMAKE_CXX_COMPILER_LAUNCHER "${Python3_EXECUTABLE};${CMAKE_CURRENT_SOURCE_DIR}/scripts/msvc.py")
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "Note: including file: ")
  add_compile_options(/utf-8 /W3)
  add_compile_definitions(_USE_MATH_DEFINES NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(zlib URL https://codeload.github.com/madler/zlib/tar.gz/refs/tags/v1.3.1 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(zlib)
add_library(ZLIB::ZLIB ALIAS zlibstatic)
set(ZLIB_INCLUDE_DIR "${zlib_SOURCE_DIR};${zlib_BINARY_DIR}" CACHE STRING "" FORCE)
set(ZLIB_LIBRARY zlibstatic CACHE STRING "" FORCE)
set(tiff-tools OFF CACHE BOOL "" FORCE)
set(tiff-tests OFF CACHE BOOL "" FORCE)
set(tiff-contrib OFF CACHE BOOL "" FORCE)
set(tiff-docs OFF CACHE BOOL "" FORCE)
set(ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(ENABLE_STATIC ON CACHE BOOL "" FORCE)
set(WITH_TOOLS OFF CACHE BOOL "" FORCE)
set(WITH_TESTS OFF CACHE BOOL "" FORCE)
set(WITH_SIMD OFF CACHE BOOL "" FORCE)
FetchContent_Declare(jpeg_turbo URL https://codeload.github.com/libjpeg-turbo/libjpeg-turbo/tar.gz/refs/tags/3.1.2 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_Populate(jpeg_turbo)
include(ExternalProject)
execute_process(COMMAND "${CMAKE_COMMAND}" -S "${jpeg_turbo_SOURCE_DIR}" -B "${jpeg_turbo_BINARY_DIR}"
  -G Ninja -DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE} -DENABLE_SHARED=OFF -DWITH_SIMD=OFF
  -DWITH_TOOLS=OFF -DWITH_TESTS=OFF RESULT_VARIABLE JPEG_CONFIGURED)
if(NOT JPEG_CONFIGURED EQUAL 0)
  message(FATAL_ERROR "Could not configure libjpeg-turbo")
endif()
ExternalProject_Add(jpeg_build SOURCE_DIR "${jpeg_turbo_SOURCE_DIR}" BINARY_DIR "${jpeg_turbo_BINARY_DIR}"
  CONFIGURE_COMMAND "" BUILD_COMMAND "${CMAKE_COMMAND}" --build . --target jpeg-static
  INSTALL_COMMAND "" BUILD_BYPRODUCTS "${jpeg_turbo_BINARY_DIR}/jpeg-static.lib")
add_library(jpeg-static STATIC IMPORTED GLOBAL)
set_target_properties(jpeg-static PROPERTIES IMPORTED_LOCATION "${jpeg_turbo_BINARY_DIR}/jpeg-static.lib"
  INTERFACE_INCLUDE_DIRECTORIES "${jpeg_turbo_SOURCE_DIR}/src;${jpeg_turbo_BINARY_DIR}")
add_dependencies(jpeg-static jpeg_build)
set(JPEG_INCLUDE_DIR "${jpeg_turbo_SOURCE_DIR}/src" CACHE STRING "" FORCE)
set(JPEG_LIBRARY jpeg-static CACHE STRING "" FORCE)
add_library(JPEG::JPEG ALIAS jpeg-static)
set(jpeg ON CACHE BOOL "" FORCE)
set(jbig OFF CACHE BOOL "" FORCE)
set(lzma OFF CACHE BOOL "" FORCE)
set(zstd OFF CACHE BOOL "" FORCE)
set(webp OFF CACHE BOOL "" FORCE)
FetchContent_Declare(libtiff URL https://download.osgeo.org/libtiff/tiff-4.7.0.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(libtiff)
FetchContent_Declare(lcms URL https://codeload.github.com/mm2/Little-CMS/tar.gz/refs/tags/lcms2.17 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_GetProperties(lcms)
if(NOT lcms_POPULATED)
  FetchContent_Populate(lcms)
endif()
file(GLOB LCMS_SOURCES CONFIGURE_DEPENDS "${lcms_SOURCE_DIR}/src/*.c")
add_library(lcms2 STATIC ${LCMS_SOURCES})
target_include_directories(lcms2 PUBLIC "${lcms_SOURCE_DIR}/include")
FetchContent_Declare(miniz URL https://codeload.github.com/richgel999/miniz/tar.gz/refs/tags/3.0.2 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(miniz)

set(PIXEL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../Compositor/Rendering")
add_library(compositor_pixels STATIC
  ${PIXEL_ROOT}/AdjustPixels.c ${PIXEL_ROOT}/BrushPixels.c
  ${PIXEL_ROOT}/ContentFill.c ${PIXEL_ROOT}/HealPixels.c
  ${PIXEL_ROOT}/LensPixels.c ${PIXEL_ROOT}/LevelsPixels.c
  ${PIXEL_ROOT}/NoisePixels.c ${PIXEL_ROOT}/WandPixels.c)

  src/core/EditorSession.cpp src/core/Editing.cpp src/core/ToolCommands.cpp src/ai/AIProvider.cpp src/ai/AIEditCoordinator.cpp
  src/ai/ConnectionHealthService.cpp src/ai/EmbeddedAI.cpp src/ai/EmbeddedImage.cpp src/ai/ManagedImage.cpp src/ai/OwnedComfy.cpp src/ai/LocalVision.cpp src/io/ImageCodec.cpp src/io/PhotoshopCodec.cpp
  src/io/ProjectStore.cpp src/io/VersionStore.cpp src/io/VersionQueue.cpp src/io/DocumentLoader.cpp)

target_compile_definitions(compositor_core PRIVATE COMPOSITOR_LOCAL_AI_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}")
set(ONNX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.tools/onnxruntime/onnxruntime-win-x64-1.20.1")
if(EXISTS "${ONNX_ROOT}/include/onnxruntime_cxx_api.h")

  target_include_directories(compositor_core PRIVATE "${ONNX_ROOT}/include")
  target_link_libraries(compositor_core PUBLIC "${ONNX_ROOT}/lib/onnxruntime.lib")
  file(COPY "${ONNX_ROOT}/lib/onnxruntime.dll" DESTINATION "${CMAKE_BINARY_DIR}")
endif()

  compositor_pixels tiff lcms2 miniz zlibstatic Qt6::Network advapi32 dxgi)

  src/ui/CanvasView.cpp src/ui/AdjustmentPanel.cpp src/ui/CurveEditor.cpp src/ui/VersionDialog.cpp
  src/render/D3DCanvas.cpp)
